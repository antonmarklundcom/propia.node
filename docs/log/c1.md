## Hub conversion

Ran this one-off Node script via stdin from the repo root (no script file or
npm script added):

```js
const sharp = require('sharp');
(async () => {
  for (const [source, target] of [
    ['casa-premium-asuncion-atardecer', 'hub-venta'],
    ['living-moderno-vista-rio-asuncion', 'hub-alquiler'],
  ]) {
    const path = `public/img/${target}.webp`;
    await sharp(`public/img/premium/${source}-1920.webp`)
      .resize({ width: 1920 }).webp({ quality: 78 }).toFile(path);
    const { format, width, height } = await sharp(path).metadata();
    console.log({ path, format, width, height, quality: 78 });
  }
})().catch(error => { console.error(error); process.exitCode = 1; });
```
